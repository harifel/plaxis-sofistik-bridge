+prog template
$-----------------------------------------------------------------------
$ Step 1: Initialize variables
$-----------------------------------------------------------------------
SEIT -1 Nzei 9999 Form 4 $ delete head of the list
LET#n 0 $ counter for nodes
LET#CDB_LEN 0
LET#CDB_IER 0
@KEY NODE 0 $ select all nodes
$-----------------------------------------------------------------------
$ Step 2: Extract node coordinates
$-----------------------------------------------------------------------
LOOP 1000 $ loop over nodes (max 1000000)
LOOP 1000
LET#helvec(#n) @nr $ node number
LET#xvec(#n) @XYZ $ x-coordinate
LET#yvec(#n) @XYZ(1) $ y-coordinate
LET#zvec(#n) @XYZ(2) $ z-coordinate
LET#n #n+1 $ increment counter
ENDLOOP #CDB_IER<2
ENDLOOP #CDB_IER<2
LET#n #n-1 $ adjust final count
$-----------------------------------------------------------------------
$ Step 3: Filter nodes for foundation slab
$-----------------------------------------------------------------------
LET#zz -0.00 $ slab elevation
LET#a 0
LET#aa 0
LOOP #n
IF ((#zvec(#a)>#zz-.0001)&(#zvec(#a)<#zz+.0001)) &
((#xvec(#a)>=-.0001)&(#xvec(#a)<=17.5)) &
((#yvec(#a)>=-.0001)&(#yvec(#a)<=17.5))
LET#nodvec(#aa) #helvec(#a) $ save filtered node
@KEY N_DISPC 10 #nodvec(#aa) $ extract displacement for node
LET#pzvec(#aa) @PZ $ get Z displacement
TXA #(#helvec(#a),10.0) $ write results: node number
#(#xvec(#a),15.9) $ x-coordinate
#(#yvec(#a),15.9) $ y-coordinate
#(#zvec(#a),15.9) $ z-coordinate
#(#pzvec(#aa),10.6) $ displacement in z
LET#aa #aa+1
ENDIF
LET#a #a+1
ENDLOOP
$-----------------------------------------------------------------------
$ Step 4: Export support forces to file
$-----------------------------------------------------------------------
ENDE
+sys copy $(projekt).lst SOFISTIK_support_forces.txt
